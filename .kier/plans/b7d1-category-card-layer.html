/** B2d2b2: overlays the card's header. Cancel takes focus, so Enter never deletes by accident. */
const BoardWorkspaceCategoryDeleteConfirm: React.FC<Props> = ({ name, onCancel, onConfirm }) => {
  const ref = useRef<HTMLDivElement>(null)
  const cancelRef = useRef<HTMLButtonElement>(null)

  useEffect(() => {
    cancelRef.current?.focus({ preventScroll: true })
    const onPointerDown = (evt: PointerEvent) => ref.current?.contains(evt.target as Node) || onCancel()
    const onKeyDown = (evt: KeyboardEvent) => evt.key === 'Escape' && onCancel()
    document.addEventListener('pointerdown', onPointerDown)
    document.addEventListener('keydown', onKeyDown)
    return () => { /* remove both */ }
  }, [onCancel])

  return (
    // 1px inside the rim; runs 12px under the tray so red fills its rounded corners.
    <div ref={ref} role="group" aria-label={`Delete ${name}?`}
      className="absolute inset-x-px top-px -bottom-3 isolate flex items-center gap-2 rounded-t-xl bg-zinc-800 pr-1.5 pb-3 pl-3 before:absolute before:inset-0 before:-z-10 before:rounded-t-xl before:bg-linear-to-r before:from-red-500/35 before:via-red-500/10 before:via-45% before:to-transparent before:to-80% before:animate-confirm-fade motion-reduce:before:animate-none">
      <Icon icon={TrashIcon} size={16} className="text-red-400 animate-trash-tip motion-reduce:animate-none" />
      <span className="animate-confirm-rise motion-reduce:animate-none">
        <span className="text-zinc-400">Delete</span> <span className="font-semibold">{name}</span>?
      </span>
      <span className="ml-auto flex gap-1.5 animate-confirm-rise-late motion-reduce:animate-none">
        <button ref={cancelRef} onClick={onCancel} className="h-6 rounded-md bg-zinc-700 px-2 text-xs font-bold hover:bg-zinc-600">Cancel</button>
        <button onClick={onConfirm} className="h-6 rounded-md bg-red-600 px-2 text-xs font-bold hover:bg-red-500">Delete</button>
      </span>
    </div>
  )
}
