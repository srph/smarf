/** Create an "Untitled" board laid out by a start. Resolves to the board, categories and all. */
const useCreateBoardMutation = () => {
  const queryClient = useQueryClient()

  return useMutation({
    // The whole start, so the mutation's variables tell the tray which one is saving
    mutationFn: async (start: ResolvedStart) => {
      const body: CreateBoardVariables = {
        name: 'Untitled',
        // Starts carry whole heroes (previews need thumbnails); the API takes ids and orders
        categories: start.categories.map(({ heroes, ...category }) => ({
          ...category,
          heroes: heroes.map((hero, i) => ({ id: hero.id, order: (i + 1) * ORDER_BUFFER }))
        }))
      }
      return (await http<{ board: Board }>('boards', { method: 'POST', body })).board
    },
    onSuccess: (board) => {
      // /b/:id renders from this instead of waiting on GET /boards/:id
      queryClient.setQueryData(boardKeys.detail(board.id), board)
      return queryClient.invalidateQueries({ queryKey: boardKeys.list() })
    }
  })
}

// The tray's fold. Navigation waits for it, so the dock picks the bar up at rest, not mid-fold.
const FOLD_MS = 500

const { mutateAsync: createBoard, status: createStatus, variables: chosen } = useCreateBoardMutation()
// 'success' stays 'saving': the route is about to unmount
const phase = createStatus === 'idle' ? 'choosing' : createStatus === 'error' ? 'failed' : 'saving'

// Set when /b/new unmounts, so a save that finishes after they've left doesn't pull them back
const leftRef = useRef(false)
useEffect(() => {
  leftRef.current = false
  return () => void (leftRef.current = true)
}, [])

const choose = async (start: ResolvedStart) => {
  try {
    const [board] = await Promise.all([createBoard(start), wait(reduceMotion ? 0 : FOLD_MS)])
    if (leftRef.current) return
    const state: NewBoardHandoffState = { newBoard: { start: start.name } }
    // replace: Back from the new board goes where New Board was clicked, not to /b/new
    navigate(`/b/${board.id}`, { replace: true, state })
  } catch {
    // phase is now 'failed': the tray unfolds with its error, and every start can be chosen again
  }
}

/** What /b/new puts in router state when it lands on the board it just created. */
interface NewBoardHandoffState { newBoard: { start: string } }

const NEW_BOARD_DOCK = 'new-board-dock'

/**
 * The dock's arrival from /b/new: 'sliding' from the "Creating" bar, then 'swapping' as "Creating …" blurs into
 * the row, then null. `arrived` holds for the toolbar's life, so the dock's rise never plays afterwards.
 */
const useNewBoardHandoff = () => {
  const location = useLocation()
  const navigate = useNavigate()
  const [start] = useState(() => (location.state as Partial<NewBoardHandoffState> | null)?.newBoard?.start ?? null)
  const [stage, setStage] = useState<'sliding' | 'swapping' | null>(start ? 'sliding' : null)

  // No layout to animate (reduced motion, jsdom) means no onLayoutAnimationComplete: time out instead
  useEffect(() => { /* 'sliding' → 'swapping' after SLIDE_MS + 100 */ }, [stage])
  useEffect(() => { /* 'swapping' → null after 250ms; then clear router state so Back or refresh can't replay it */ }, [stage])

  return { arrived: start !== null, start, stage, slid: () => setStage((s) => (s === 'sliding' ? 'swapping' : s)) }
}

Decisions

How does /b/:id get the new board?
POST returns it whole; seed the detail cache
Navigate and let GET /boards/:id fetch it
No spinner between choosing and the board.
How does the bar become the dock across routes?
Motion layoutId on both
One bar in a shared /b layout route
No slide: cut to the dock
The seeded cache renders /b/:id in the same commit.
When does /b/new navigate?
After both the save and the fold
As soon as the save returns
A fast save would cut the fold mid-way.
How does the dock know it's arriving?
Router state, cleared after the swap
A module-level one-shot flag
Explicit; cleared, so it never replays.

Set only while sliding.
