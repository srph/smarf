interface Props {
  value: CategoryColor | null
  onPreview: (color: CategoryColor | null | undefined) => void // undefined: back to the saved colour
  onSelect: (color: CategoryColor | null) => void
  onCancel: () => void
}

/**
 * Picks a category's colour over the card's header, the way the delete confirm asks its question: same inset, same
 * rise. Hovering or focusing a swatch previews it on the card. Escape, the X, or a click anywhere else backs out.
 */
const BoardWorkspaceCategoryColorPicker: React.FC<Props> = ({ value, onPreview, onSelect, onCancel }) => {
  // Focus starts on the current colour. Escape and outside pointerdown: the delete confirm's two effects.
  return (
    <div ref={ref} role="radiogroup" aria-label="Category colour" onPointerLeave={() => onPreview(undefined)}
      className="absolute inset-x-px top-px -bottom-3 isolate flex items-center gap-1.5 bg-zinc-800 pr-1 pb-3 pl-2.5 ...">
      {[null, ...CATEGORY_COLORS.map((c) => c.key)].map((color) => (
        <button key={color ?? 'none'} type="button" role="radio" aria-checked={color === value}
          aria-label={color ? nameOf(color) : 'None'} title={...}
          onPointerEnter={() => onPreview(color)} onFocus={() => onPreview(color)} onClick={() => onSelect(color)}
          className={cn('grid size-5 place-items-center rounded-full hover:ring-2 hover:ring-primary-500 focus-visible:ring-focus',
            color ? cn('bg-category-metal', categoryColorClass[color]) : 'inset-ring-2 inset-ring-zinc-500')}>
          {color === value && color && <Icon icon={CheckIcon} size={12} />} {/* 16/solid check */}
        </button>
      ))}
      <button type="button" title="Cancel" aria-label="Cancel" onClick={onCancel} className="ml-auto grid size-7 ...">
        <Icon icon={XMarkIcon} size={16} />
      </button>
    </div>
  )
}

Decisions

Trigger icon

  • Swatch your pick
  • Paint brush
  • Eye dropper

Heroicons SwatchIcon at 16px, neutral like the trash.

Red in the palette

  • Keep red your pick
  • Swap for rose
  • Drop it

The confirm still differs by its question, buttons, grey heroes and red rim.

Extras

  • Pre-coloured starts your pick
  • Showcase film: not in this change

The film's tier list will lag the app until it's updated.

What the column holds

  • A key, red pick
  • A hex value

Keys let the look change later without touching rows, and validate with one Rule::in.

How the save is sent

  • Extend useUpdateCategoryMutation pick
  • A second hook on the same endpoint

One hook per request; the endpoint already takes partial fields.

How a colour reaches CSS

  • One variable, two utilities pick
  • Literal from/via classes per colour

Eight map entries instead of thirty-two, and the preview swaps a single class.

F Tier in the start

  • None pick
  • Violet

Grey reads as the bottom; say if you want a colour there.

Risks

medium Tailwind drops a colour variable

Check the built CSS defines all eight --color-*-500; if not, list them in @theme.

medium Two overlays on one header

Opening the picker or the confirm closes the other; each already backs out on an outside pointerdown.

low Colour lists drift apart

The API test loops Category::COLORS; the UI type is the same eight keys, cross-referenced in comments.

low Preview left on after closing

Closing always resets previewColor to undefined, in one closeColorPicker.

low Fake layout finds the tray by position

The picker mounts inside the header, so the header and Resizable stay the card's first two children.
